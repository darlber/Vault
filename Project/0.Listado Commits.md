async function getCommitFinalText(file) {
  let content = await dv.io.load(file.path);
  const lines = content.split("\n");
  let capture = false;
  let commitLines = [];

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i].trim();
    // Encabezado nivel 1 exacto
    if (line === "# Commit Final") {
      capture = true;
      i++;
      while (i < lines.length && !lines[i].startsWith("# ")) { // Para hasta el siguiente header nivel 1
        commitLines.push(lines[i]);
        i++;
      }
      break;
    }
  }

  return commitLines.join("\n").trim();
}

const pages = dv.pages('"Project"');
const rows = [];

for (let page of pages) {
  const commitText = await getCommitFinalText(page.file);
  if (commitText.length > 0) {
    const dateMatch = page.file.name.match(/\d{4}-\d{2}-\d{2}/);
    const date = dateMatch ? dateMatch[0] : "0000-00-00";
    rows.push({
      link: page.file.link,
      date,
      commit: commitText
    });
  }
}

rows.sort((a, b) => a.date.localeCompare(b.date));

dv.table(
  ["Nota", "Commit Final"],
  rows.map(r => [r.link, r.commit.length > 200 ? r.commit.substring(0, 200) + "..." : r.commit])
);
