package com.example.exerlog.ui.home.components  
  
import androidx.compose.foundation.clickable  
import androidx.compose.foundation.combinedClickable  
import androidx.compose.foundation.layout.*  
import androidx.compose.material3.*  
import androidx.compose.runtime.Composable  
import androidx.compose.runtime.derivedStateOf  
import androidx.compose.runtime.getValue  
import androidx.compose.runtime.remember  
import androidx.compose.ui.Alignment  
import androidx.compose.ui.Modifier  
import androidx.compose.ui.tooling.preview.Preview  
import androidx.compose.ui.unit.dp  
import com.example.exerlog.db.entities.Session  
import com.example.exerlog.ui.SessionWrapper  
import com.example.exerlog.ui.home.HomeEvent  
  
import java.time.LocalDateTime  
  
@Composable  
fun SessionCard(  
    sessionWrapper: SessionWrapper,  
    onClick: () -> Unit,  
    onLongClick: () -> Unit  
) {  
    val muscleTitle by remember {  
        derivedStateOf {  
            sessionWrapper.muscleGroups.firstOrNull()?: "NO MUSCLE"  
        }  
    }  
  
    val muscleSubtitle by remember {  
        derivedStateOf {  
            sessionWrapper.muscleGroups.drop(1)  
                .take(3)  
                .joinToString(", ")  
        }  
    }  
  
  
  
    Card(  
        modifier = Modifier  
            .fillMaxWidth()  
            .padding(horizontal = 16.dp, vertical = 8.dp)  
            .combinedClickable(  
                onClick = { onClick() },  
                onLongClick = { onLongClick() }  
            ),  
                // Estilo de la tarjeta  
        shape = MaterialTheme.shapes.small  
  
    ) {  
        Row(  
            modifier = Modifier  
                .fillMaxWidth()  
                .padding(16.dp)  
        ) {  
            // Fecha a la izquierda  
            SessionDate(  
                session = sessionWrapper.session,  
                modifier = Modifier.padding(end = 16.dp).align(Alignment.CenterVertically)  
            )  
            // Info principal  
            Column(  
                modifier = Modifier.weight(1f),  
                verticalArrangement = Arrangement.Center  
            ) {  
                Text(  
                    text = muscleTitle,  
                    style = MaterialTheme.typography.headlineSmall  
                )  
                if (muscleSubtitle.isNotEmpty()) {  
                    Text(  
                        text = muscleSubtitle,  
                        style = MaterialTheme.typography.bodySmall,  
                        color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.7f)  
                    )  
                }  
            }  
        }  
    }  
}

# AppNavHost.kt
```kotlin
package com.example.exerlog.ui  
  
import androidx.compose.runtime.Composable  
import androidx.navigation.NavHostController  
import androidx.navigation.compose.NavHost  
import androidx.navigation.compose.composable  
import com.example.exerlog.core.Routes  
import com.example.exerlog.ui.home.HomeScreen  
import com.example.exerlog.ui.settings.SettingsScreen  
import com.example.exerlog.utils.UiEvent  
  
@Composable  
fun AppNavHost(  
    navController: NavHostController  
) {  
  
    NavHost(  
        navController = navController,  
        startDestination = Routes.HOME  
    ) {  
  
        composable(Routes.HOME) {  
            HomeScreen(  
                onNavigate = { navController.navigationEvent(event = it) },  
            )  
        }  
        composable(Routes.SETTINGS) {  
            SettingsScreen()  
        }  
  
    }  
}  
  
private fun NavHostController.navigationEvent(event: UiEvent.Navigate) {  
    navigate(event.route) {  
        if (event.popBackStack) currentDestination?.route?.let { popUpTo(it) { inclusive = true } }  
        launchSingleTop = true  
    }  
}
```

# Commit Final

Feat: Implement settings screen and session deletion  
  
This commit introduces the `SettingsScreen` with basic functionality for database export, import, and clearing. It also adds the ability to delete sessions from the home screen.  
  
**Settings:**  
- **`SettingsScreen.kt`:**  
    - New composable screen for settings.  
    - Provides buttons for "Export Database", "Import Database", and "Delete Database".  
    - Uses `rememberLauncherForActivityResult` for file creation (export) and content selection (import).  
    - Communicates with `SettingsViewModel` via `SettingsEvent`.  
- **`SettingsViewModel.kt`:**  
    - New ViewModel to handle settings logic.  
    - Placeholder logic for database export/import (currently commented out).  
    - Implements `ClearDatabase` event to call `repo.clearDatabase()`.  
- **`SettingsEvent.kt`:**  
    - New sealed class defining events for the settings screen: `ExportDatabase`, `ImportDatabase`, `CreateFile`, `ClearDatabase`.  
  
**Home Screen & Session Management:**  
- **`HomeViewModel.kt`:**  
    - Added `DeleteSessionRequested` and `ConfirmDeleteSession` events to `HomeEvent`.  
    - Manages a `sessionToDelete` state to control the delete confirmation dialog.  
    - `ConfirmDeleteSession` calls `repo.deleteSessionById(sessionId)`.  
- **`HomeScreen.kt`:**  
    - Displays an `AlertDialog` to confirm session deletion when `viewModel.sessionToDelete` is not null.  
    - `SessionCard` now triggers `HomeEvent.DeleteSessionRequested` on long click.  
- **`SessionCard.kt`:**  
    - Now uses `combinedClickable` to handle both `onClick` and `onLongClick`.  
    - Improved display of muscle groups: shows the primary muscle as a headline and subsequent muscles as a subtitle.  
- **`ExerDAO.kt`:** Added `deleteSessionById(sessionId: Long)` method.  
- **`ExerRepository.kt` & `ExerRepositoryImpl.kt`:**  
    - Added `deleteSessionById(sessionId: Long)` to the interface and implementation.  
    - `ExerRepositoryImpl.clearDatabase()` now also clears `sqlite_sequence`.  
- **`ExerciseExtension.kt`:** `sortedListOfMuscleGroups` now only considers `primaryMuscles`.  
  
**Navigation & DI:**  
- **`AppNavHost.kt`:**  
    - Added a route for `SETTINGS` to navigate to `SettingsScreen`.  
    - Improved `navigationEvent` to handle `popBackStack`.  
- **`DatabaseModule.kt`:** `provideRepositoryImpl` now also injects `GymDatabase` into `ExerRepositoryImpl`.  
  
**Other:**  
- Updated deployment target in `.idea/deploymentTargetSelector.xml`.
