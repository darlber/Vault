# SessionDate.kt
```kotlin
package com.example.exerlog.ui.home.components  
  
import androidx.compose.foundation.layout.Column  
import androidx.compose.material3.MaterialTheme  
import androidx.compose.material3.Text  
import androidx.compose.runtime.Composable  
import androidx.compose.runtime.derivedStateOf  
import androidx.compose.runtime.getValue  
import androidx.compose.runtime.remember  
import androidx.compose.ui.Alignment  
import androidx.compose.ui.Modifier  
import androidx.compose.ui.platform.LocalConfiguration  
import androidx.compose.ui.text.font.FontWeight  
import com.example.exerlog.db.entities.Session  
import java.time.format.TextStyle  
  
@Composable  
fun SessionDate(  
    session: Session,  
    modifier: Modifier = Modifier  
) {  
    val configuration = LocalConfiguration.current  
    val locale = configuration.locales[0]  
  
    val month by remember {  
        derivedStateOf {  
            session.start.month.getDisplayName(TextStyle.SHORT, locale)  
        }  
    }  
    val day by remember { derivedStateOf { session.start.dayOfMonth.toString() } }  
  
    Column(  
        horizontalAlignment = Alignment.CenterHorizontally,  
        modifier = modifier  
    ) {  
        DateText(text = month)  
        DateText(text = day)  
    }  
}  
  
@Composable  
fun DateText(text: String) {  
    Text(  
        text = text,  
        style = MaterialTheme.typography.bodyMedium,  
        fontWeight = FontWeight.SemiBold  
    )  
}
```

```kotlin
import kotlinx.coroutines.flow.Flow  
import kotlinx.coroutines.flow.flatMapLatest  
import kotlinx.coroutines.flow.mapNotNull  
import timber.log.Timber  
  
  
class GymRepository(  
    private val dao: GymDAO  
) {  
  
    fun getSessionById(sessionId: Long) = dao.getSessionById(sessionId)  
  
    fun getAllSessions() = dao.getAllSessions()  
  
    fun getAllSets() = dao.getAllSets()

    @OptIn(ExperimentalCoroutinesApi::class)

        return session.flatMapLatest {  
            dao.getExercisesForSession(it.sessionId)  
        }  
    }  
  
    fun getExercisesForSession(session: Session): Flow<List<SessionExerciseWithExercise>> {  
        Timber.d("Retrieving exercises for session: $session")  
        return dao.getExercisesForSession(session.sessionId)

    fun getSetsForExercise(sessionExerciseId: Long) = dao.getSetsForExercise(sessionExerciseId)

        val list = dao.getMuscleGroupsForSession(session.sessionId).mapNotNull {

            try {  
                turnTargetIntoMuscleGroups(it)  
            } catch (_: Exception) {  
                Timber.d("Error when converting target.")  
                emptyList()  
            }  
        }  
        return list  
    }  
  
    suspend fun insertExercise(exercise: Exercise) = dao.insertExercise(exercise)  
  
    suspend fun insertSession(session: Session) = dao.insertSession(session)  
  
    suspend fun removeSession(session: Session) = dao.removeSession(session)  
  
    suspend fun updateSession(session: Session) = dao.updateSession(session)  
  
    suspend fun insertSessionExercise(sessionExercise: SessionExercise) =  
        dao.insertSessionExercise(sessionExercise)  
  
    suspend fun removeSessionExercise(sessionExercise: SessionExercise) =  
        dao.removeSessionExercise(sessionExercise)  
  
    suspend fun insertSet(gymSet: GymSet) = dao.insertSet(gymSet)  
  
    suspend fun updateSet(set: GymSet) = dao.updateSet(set)  
    suspend fun deleteSet(set: GymSet) = dao.deleteSet(set)  
  
    suspend fun createSet(sessionExercise: SessionExercise) =  
        dao.insertSet(GymSet(parentSessionExerciseId = sessionExercise.sessionExerciseId))  
  
    fun getDatabaseModel() = DatabaseModel(  
        sessions = dao.getSessionList(),  
        exercises = dao.getExerciseList(),  
        sessionExercises = dao.getSessionExerciseList(),  
        sets = dao.getSetList()  
    )  
  
    suspend fun clearDatabase() {  
        dao.clearSessions()  
        dao.clearSessionExercises()  
        dao.clearExercises()  
        dao.clearSets()  
    }  
}
```

# Commit Final 
Refactor: Migrate constants to enums and core package  
  
This commit refactors the constant objects `Equipment`, `MuscleGroup`, and `TipoSet` into enums and moves them to the `core` package. It also introduces `Entities` and `Routes` objects for better organization.  
  
**Key Changes:**  
  
- **Constant Objects to Enums:**  
    - `Equipment.kt`: Converted from an object to an enum, providing type safety and better organization.  
    - `MuscleGroup.kt`: Converted from an object to an enum.  
    - `TipoSet.kt`: Remains an object but moved to the `core` package for consistency.  
- **New Core Objects:**  
    - `Entities.kt`: Introduced to store string constants for database table names, improving maintainability and reducing magic strings.  
    - `Routes.kt`: Introduced to store navigation route constants.  
- **DAO Update:**  
    - `ExerDAO.kt`: Updated to use the constants from `Entities.kt` for table names in queries.  
- **UI State and Components:**  
    - `UiState.kt`: New file defining data classes for UI state (`SessionWrapper`, `ExerciseWrapper`, `TimerState`, `DatabaseModel`).  
    - `Home.kt`: New composable screen for the home view.  
    - `HomeBottomBar.kt`: New composable for the bottom navigation bar on the home screen.  
    - `SessionCard.kt`: New composable for displaying session information as a card.  
    - `SessionDate.kt`: New composable for displaying the date of a session.  
- **Repository:**  
    - `ExerRepository.kt`: New interface defining the repository contract.  
    - `ExerRepositoryImpl.kt`: New implementation of the `ExerRepository` interface.  
- **Utility:**  
    - `Event.kt`: New empty interface, likely a placeholder for future event handling.  
- **Dependencies:**  
    - Added `androidx.compose.material3:material3` dependency.  
- **Project Configuration:**  
    - Added VCS mapping for Git.  
    - Updated `.idea/misc.xml`.  
  
**Impact:**  
  
- Improved code organization and maintainability by centralizing constants and using enums.  
- Enhanced type safety with the use of enums.  
- Introduced foundational UI components and state management classes for the home screen.  
- Established a repository pattern for data access.
