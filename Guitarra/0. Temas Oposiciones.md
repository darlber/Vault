```dataviewjs
const folder = "Guitarra"; // Make sure this is the *exact folder name* in quotes
const pages = dv.pages(`"${folder}"`);

for (let page of pages.sort(p => p.file.name)) {
    const content = await dv.io.load(page.file.path);
    const match = content.match(/^# (.+)$/m);
    const heading = match ? match[1] : "*No heading found*";
    dv.paragraph(`- [[${page.file.name}]] — ${heading}`);
}
```
